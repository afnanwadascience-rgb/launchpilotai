import { execFileSync } from 'node:child_process';
import { mkdir, readFile, rm, writeFile } from 'node:fs/promises';
import { dirname, join } from 'node:path';
import { fileURLToPath } from 'node:url';

const root = dirname(fileURLToPath(import.meta.url));
const output = join(root, 'dist');

// Prisma needs URL-shaped datasource values while generating its client, but it
// does not connect to the database during generation. Placeholders are passed
// to this child process only when real values have not been configured; they
// are never written to .env or to the deployment environment.
const env = { ...process.env };
const fallbackDatabaseUrl = 'postgresql://placeholder:placeholder@127.0.0.1:5432/launchpilot_placeholder?schema=public';
env.DATABASE_URL ||= fallbackDatabaseUrl;
env.DIRECT_URL ||= env.DATABASE_URL;
const prismaCommand = join(root, 'node_modules', '.bin', process.platform === 'win32' ? 'prisma.cmd' : 'prisma');
execFileSync(prismaCommand, ['generate'], {
  cwd: root,
  env,
  stdio: 'inherit',
  shell: process.platform === 'win32'
});

// Load only the public checkout URL from a local .env file. Vercel build
// environment variables take precedence. No secrets are inserted into HTML.
function parseEnvFile(text) {
  const values = {};
  for (const line of text.split(/\r?\n/)) {
    const match = line.match(/^\s*([A-Z0-9_]+)\s*=\s*(.*?)\s*$/);
    if (!match || match[1] !== 'WHOP_PRO_CHECKOUT_URL') continue;
    let value = match[2];
    if ((value.startsWith('"') && value.endsWith('"')) || (value.startsWith("'") && value.endsWith("'"))) value = value.slice(1, -1);
    values[match[1]] = value;
  }
  return values;
}

const envText = await readFile(join(root, '.env'), 'utf8').catch(() => '');
const localEnv = parseEnvFile(envText);
const checkoutUrl = (process.env.WHOP_PRO_CHECKOUT_URL || localEnv.WHOP_PRO_CHECKOUT_URL || '').trim();
if (checkoutUrl) {
  let parsedCheckoutUrl;
  try { parsedCheckoutUrl = new URL(checkoutUrl); } catch { throw new Error('WHOP_PRO_CHECKOUT_URL must be a valid HTTPS Whop checkout URL.'); }
  if (parsedCheckoutUrl.protocol !== 'https:' || !(parsedCheckoutUrl.hostname === 'whop.com' || parsedCheckoutUrl.hostname.endsWith('.whop.com'))) {
    throw new Error('WHOP_PRO_CHECKOUT_URL must use HTTPS and a whop.com domain.');
  }
}

// The authoring file stays at the repository root. The public checkout URL is
// injected during build, then only dist/index.html is served as the frontend.
await rm(output, { recursive: true, force: true });
await mkdir(output, { recursive: true });
const html = await readFile(join(root, 'index.html'), 'utf8');
const marker = "const CHECKOUT_URL = '__WHOP_PRO_CHECKOUT_URL__';";
if (!html.includes(marker)) throw new Error('Checkout URL marker is missing from index.html.');
const builtHtml = html.replace(marker, `const CHECKOUT_URL = ${JSON.stringify(checkoutUrl)};`);
await writeFile(join(output, 'index.html'), builtHtml, 'utf8');
console.log(checkoutUrl
  ? 'LaunchPilot AI frontend built to dist/index.html with a configured Whop checkout URL.'
  : 'LaunchPilot AI frontend built to dist/index.html. Pro checkout buttons are safely disabled until WHOP_PRO_CHECKOUT_URL is configured.');

import { randomUUID } from 'node:crypto';
import { Prisma } from '@prisma/client';
import { aiRequestSchema } from '../../lib/validators';
import { isAiTool, TOOL_DETAILS } from '../../lib/constants';
import { generateWithGroq } from '../../lib/ai';
import { failGeneration, getPlanContext, reserveGeneration } from '../../lib/usage';
import { requireApiUser } from '../../lib/auth';
import { assertSameOrigin, jsonError, readJsonLimited } from '../../lib/http';
import { consumeRateLimit } from '../../lib/rate-limit';
import { db } from '../../lib/db';
import { canAccessTool } from '../../lib/policy';

export const runtime = 'nodejs';
export const maxDuration = 60;
export async function POST(request: Request) {
  if (!assertSameOrigin(request)) return jsonError('Request origin could not be validated.', 403);
  const { user, response } = await requireApiUser(request);
  if (!user) return response;
  try {
    if (!(await consumeRateLimit(`ai:${user.id}`, 12, 60 * 60 * 1000))) return jsonError('You have made too many AI requests. Please try again later.', 429, 'RATE_LIMITED');
    const raw = await readJsonLimited(request, 18_000);
    const parsed = aiRequestSchema.safeParse(raw);
    if (!parsed.success) return jsonError(parsed.error.issues[0]?.message || 'Please check your inputs.', 400, 'VALIDATION_ERROR');
    const { tool, input } = parsed.data;
    if (!isAiTool(tool)) return jsonError('Unknown tool.', 400);
    const planContext = await getPlanContext(user.id);
    if (!canAccessTool(planContext.plan, tool)) return jsonError('This feature is available on Pro. Verify your membership or upgrade to continue.', 403, 'PRO_REQUIRED');

    const projectId = 'projectId' in parsed.data ? (parsed.data.projectId as string | null | undefined) : undefined;
    if (projectId) {
      const owned = await db.project.findFirst({ where: { id: projectId, userId: user.id }, select: { id: true } });
      if (!owned) return jsonError('That project was not found in your account.', 404);
    }
    const headerKey = request.headers.get('idempotency-key');
    const requestKey = (headerKey && /^[a-zA-Z0-9._:-]{8,120}$/.test(headerKey)) ? headerKey : randomUUID();
    const reservation = await reserveGeneration(user.id, requestKey, tool);
    if (reservation.mode === 'limit') return Response.json({ error: `You've used your ${reservation.limit} ${reservation.plan === 'PRO' ? 'Pro' : 'Free'} AI generations.`, code: 'QUOTA_EXCEEDED' }, { status: 402 });
    if (reservation.mode === 'pending') return jsonError('This generation is already running. Wait a moment before retrying.', 409, 'REQUEST_PENDING');
    if (reservation.mode === 'cached') {
      const cached = reservation.response as { result?: string; contentId?: string } | null;
      const quota = await getPlanContext(user.id);
      return Response.json({ ok: true, result: cached?.result || '', contentId: cached?.contentId, quota, cached: true });
    }

    try {
      const result = await generateWithGroq(tool, input as unknown as Record<string, unknown>);
      const title = `${TOOL_DETAILS[tool].title} Â· ${new Date().toLocaleDateString('en-US', { month: 'short', day: 'numeric' })}`;
      const content = await db.$transaction(async (tx) => {
        const saved = await tx.generatedContent.create({ data: { userId: user.id, projectId: projectId || null, type: tool, title, body: result, metadata: input as Prisma.InputJsonValue } });
        const responsePayload = { result, contentId: saved.id };
        await tx.usageRecord.update({ where: { userId_requestKey: { userId: user.id, requestKey } }, data: { status: 'COMPLETE', response: responsePayload, completedAt: new Date() } });
        return saved;
      }, { isolationLevel: 'Serializable' });
      const quota = await getPlanContext(user.id);
      return Response.json({ ok: true, result, contentId: content.id, quota, cached: false });
    } catch (error) {
      await failGeneration(user.id, requestKey).catch(() => undefined);
      if (error instanceof Error && error.message === 'GROQ_NOT_CONFIGURED') return jsonError('AI generation is not configured yet. Add GROQ_API_KEY to the server environment.', 503, 'AI_NOT_CONFIGURED');
      if (error instanceof Error && error.message === 'EMPTY_AI_RESPONSE') return jsonError('The AI service returned an empty response. Please try again.', 502);
      console.error('ai_generation_failed', error instanceof Error ? error.message : 'unknown');
      return jsonError('AI generation failed. Your quota was not consumed; please try again.', 502, 'AI_FAILED');
    }
  } catch (error) {
    if (error instanceof Error && error.message === 'REQUEST_TOO_LARGE') return jsonError('Request is too large.', 413);
    if (error instanceof Error && error.message === 'INVALID_JSON') return jsonError('Please submit valid form data.', 400);
    if (error instanceof Error && error.message.includes('P2034')) return jsonError('Another generation is being processed. Try again shortly.', 409);
    console.error('ai_request_failed', error instanceof Error ? error.message : 'unknown');
    return jsonError('Unable to process the request right now.', 503);
  }
}

import { createHash } from 'node:crypto';
import bcrypt from 'bcryptjs';
import { loginSchema } from '../../lib/validators';
import { assertSameOrigin, getClientIp, jsonError, readJsonLimited } from '../../lib/http';
import { consumeRateLimit } from '../../lib/rate-limit';
import { db } from '../../lib/db';
import { createSession } from '../../lib/auth';

export const runtime = 'nodejs';
export async function POST(request: Request) {
  if (!assertSameOrigin(request)) return jsonError('Request origin could not be validated.', 403);
  try {
    const parsed = loginSchema.safeParse(await readJsonLimited(request, 8_000));
    if (!parsed.success) return jsonError('Enter a valid email and password.', 400);
    const { email, password } = parsed.data;
    const emailKey = createHash('sha256').update(email).digest('hex').slice(0, 32);
    const allowed = await consumeRateLimit(`login:${getClientIp(request)}:${emailKey}`, 8, 15 * 60 * 1000);
    if (!allowed) return jsonError('Too many sign-in attempts. Please wait and try again.', 429);
    const user = await db.user.findUnique({ where: { email }, select: { id: true, email: true, name: true, passwordHash: true, createdAt: true } });
    const matches = user?.passwordHash ? await bcrypt.compare(password, user.passwordHash) : false;
    if (!user || !matches) return jsonError('Email or password is incorrect.', 401);
    const cookie = await createSession(user.id);
    return Response.json({ user: { id: user.id, email: user.email, name: user.name, createdAt: user.createdAt } }, { headers: { "Set-Cookie": cookie } });
  } catch (error) {
    if (error instanceof Error && ['REQUEST_TOO_LARGE', 'INVALID_JSON'].includes(error.message)) return jsonError('Please submit valid form data.', 400);
    console.error('login_failed', error instanceof Error ? error.message : 'unknown');
    return jsonError('Sign-in is temporarily unavailable. Please try again.', 503);
  }
}

import bcrypt from 'bcryptjs';
import { registrationSchema } from '../../lib/validators';
import { assertSameOrigin, getClientIp, jsonError, readJsonLimited } from '../../lib/http';
import { consumeRateLimit } from '../../lib/rate-limit';
import { db } from '../../lib/db';
import { createSession } from '../../lib/auth';

export const runtime = 'nodejs';
export async function POST(request: Request) {
  if (!assertSameOrigin(request)) return jsonError('Request origin could not be validated.', 403);
  try {
    if (!(await consumeRateLimit(`register:${getClientIp(request)}`, 6, 60 * 60 * 1000))) return jsonError('Too many account attempts. Please try again later.', 429);
    const parsed = registrationSchema.safeParse(await readJsonLimited(request, 8_000));
    if (!parsed.success) return jsonError(parsed.error.issues[0]?.message || 'Please check the form fields.', 400, 'VALIDATION_ERROR');
    const { name, email, password } = parsed.data;
    const passwordHash = await bcrypt.hash(password, 12);
    const user = await db.user.create({ data: { name, email, passwordHash, subscription: { create: { plan: 'FREE', status: 'FREE' } } }, select: { id: true, email: true, name: true, createdAt: true } });
    const cookie = await createSession(user.id);
    return Response.json({ user }, { status: 201, headers: { "Set-Cookie": cookie } });
  } catch (error) {
    if (error && typeof error === 'object' && 'code' in error && error.code === 'P2002') return jsonError('An account with this email already exists.', 409);
    if (error instanceof Error && ['REQUEST_TOO_LARGE', 'INVALID_JSON'].includes(error.message)) return jsonError(error.message === 'REQUEST_TOO_LARGE' ? 'Request is too large.' : 'Please submit valid form data.', 400);
    console.error('register_failed', error instanceof Error ? error.message : 'unknown');
    return jsonError('Account creation is temporarily unavailable. Please check the database configuration and try again.', 503);
  }
}

import { z } from 'zod';
import { requireApiUser } from '../../lib/auth';
import { fetchWhopMembership, membershipMatchesConfiguredProduct, syncMembershipForEmail } from '../../lib/billing';
import { getPlanContext } from '../../lib/usage';
import { assertSameOrigin, jsonError, readJsonLimited } from '../../lib/http';
import { consumeRateLimit } from '../../lib/rate-limit';

const schema = z.object({ membershipId: z.string().trim().min(8).max(160) });
export const runtime = 'nodejs';
export async function POST(request: Request) {
  if (!assertSameOrigin(request)) return jsonError('Request origin could not be validated.', 403);
  const { user, response } = await requireApiUser(request); if (!user) return response;
  if (!(await consumeRateLimit(`billing-refresh:${user.id}`, 8, 60 * 60 * 1000))) return jsonError('Too many verification attempts. Please try again later.', 429);
  try {
    const parsed = schema.safeParse(await readJsonLimited(request, 3_000));
    if (!parsed.success) return jsonError('Enter a valid Whop membership ID.', 400);
    const membership = await fetchWhopMembership(parsed.data.membershipId);
    if (membership.user?.email?.toLowerCase() !== user.email.toLowerCase()) return jsonError('That membership does not match your signed-in account email.', 403);
    if (!membershipMatchesConfiguredProduct(membership)) return jsonError('This membership does not match the configured LaunchPilot Pro product and plan.', 403);
    const synced = await syncMembershipForEmail(membership);
    if (!synced) return jsonError('Membership was verified but could not be linked to this account. Confirm that the Whop account email matches your LaunchPilot account.', 403);
    const usage = await getPlanContext(user.id);
    return Response.json({ ok: true, usage, message: `Membership checked with Whop. Current plan: ${usage.plan}.` });
  } catch (error) {
    if (error instanceof Error && error.message === 'WHOP_NOT_CONFIGURED') return jsonError('Whop verification is not configured yet. Add the server-side Whop API key and product/plan IDs.', 503, 'WHOP_NOT_CONFIGURED');
    if (error instanceof Error && error.message === 'MEMBERSHIP_NOT_FOUND') return jsonError('Whop did not find that membership ID.', 404);
    if (error instanceof Error && error.message === 'INVALID_MEMBERSHIP_ID') return jsonError('That membership ID format is not valid.', 400);
    return jsonError('Could not verify membership with Whop right now. No access status was changed.', 502);
  }
}

import { projectCreateSchema } from '../lib/validators';

export const runtime = 'nodejs';
export async function GET(request: Request) {
  const user = await getCurrentUser(request);
  if (!user) return jsonError('Please sign in to view projects.', 401);
  const [projects, outputs] = await Promise.all([
    db.project.findMany({ where: { userId: user.id }, orderBy: { updatedAt: 'desc' }, include: { _count: { select: { generatedContent: true, tasks: true } } } }),
    db.generatedContent.findMany({ where: { userId: user.id }, orderBy: { createdAt: 'desc' }, take: 50, select: { id: true, title: true, type: true, body: true, projectId: true, createdAt: true } })

  return Response.json({ projects, outputs });
}

export async function POST(request: Request) {
  if (!assertSameOrigin(request)) return jsonError('Request origin could not be validated.', 403);
  const { user, response } = await requireApiUser(request); if (!user) return response;
  try {
    const parsed = projectCreateSchema.safeParse(await readJsonLimited(request, 5_000));
    if (!parsed.success) return jsonError(parsed.error.issues[0]?.message || 'Please check the project fields.', 400);
    const usage = await getPlanContext(user.id);
    const count = await db.project.count({ where: { userId: user.id } });
    const limit = usage.plan === 'PRO' ? 100 : 3;
    if (!canCreateProject(usage.plan, count)) return Response.json({ error: `Your ${usage.plan === 'PRO' ? 'Pro' : 'Free'} plan allows ${limit} saved projects.`, code: 'PROJECT_LIMIT' }, { status: 402 });
    const project = await db.project.create({ data: { userId: user.id, title: parsed.data.title, description: parsed.data.description || null } });
    return Response.json({ project }, { status: 201 });

function recordValue(record: Record<string, unknown> | undefined, key: string): unknown {
  return record && typeof record === 'object' ? record[key] : undefined;
}

export async function POST(request: Request) {
  const rawBody = await request.text();
  if (new TextEncoder().encode(rawBody).byteLength > 100_000) return jsonError('Webhook payload is too large.', 413);
  if (!verifyWhopWebhook(rawBody, request.headers)) return jsonError('Webhook signature could not be verified.', 401);
  let body: unknown;
  try { body = JSON.parse(rawBody); } catch { return jsonError('Webhook payload is not valid JSON.', 400); }
  if (!body || typeof body !== 'object' || Array.isArray(body)) return jsonError('Webhook payload shape is not valid.', 400);
  const payload = body as Record<string, unknown>;
  const data = (payload.data && typeof payload.data === 'object' ? payload.data : payload.resource && typeof payload.resource === 'object' ? payload.resource : payload) as Record<string, unknown>;
  const eventType = String(payload.type || payload.event_type || payload.event || '').slice(0, 120);
  const eventId = request.headers.get('webhook-id') || String(payload.id || payload.event_id || '') || createHash('sha256').update(rawBody).digest('hex');
  const membershipId = String(
    data.id && String(data.id).startsWith('mem_') ? data.id :
    recordValue(data.membership as Record<string, unknown> | undefined, 'id') ||
    recordValue(data, 'membership_id') ||
    recordValue(payload.membership as Record<string, unknown> | undefined, 'id') ||
    payload.membership_id || ''
  );
  try {
    const claim = await claimWebhookEvent(eventId, eventType || 'unknown', payload);
    if (claim === 'duplicate') return Response.json({ ok: true, duplicate: true });

    const membershipSignal = eventType.startsWith('membership.') || eventType === 'payment.succeeded' || eventType === 'payment.failed';
    if (membershipSignal && membershipId.startsWith('mem_')) {
      try {
        const membership = await fetchWhopMembership(membershipId);
        const linked = await syncMembershipForEmail(membership);
        if (!linked) {
          // It may be another product's event. It is valid but must not affect LaunchPilot access.
        }
      } catch (error) {
        if (eventType === 'membership.deactivated' && (error instanceof Error && error.message === 'MEMBERSHIP_NOT_FOUND')) {
          await db.subscription.updateMany({ where: { whopMembershipId: membershipId }, data: { plan: 'FREE', status: 'INACTIVE', lastVerifiedAt: new Date() } });
        } else {
          await releaseUnprocessedWebhookEvent(eventId).catch(() => undefined);
